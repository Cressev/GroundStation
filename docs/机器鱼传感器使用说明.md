# 1. IMU数据

1. 在机器鱼的ros系统中，有一个测试IMU数据发布和订阅的launch文件是：
```imu.launch
    <!-- IMU发布节点 -->
    <node pkg="TransducerM_pkg" type="TMSerial" name="imu_pub" output="screen" />

    <!-- IMU订阅节点 -->
    <!-- imu_nums_T: 订阅节点的运行时间，以imu数据发布周期数来计算，imu发布频率为50hz -->
    <!-- enable_save 是否保存数据 不区分大小写 -->
    <node pkg="fish_control" type="imu_listener.py" name="imu_listener" output="screen">
        <param name = "imu_nums_T" value="8000"/> 
        <param name = "enable_save" value="true" /> 
    </node>
```
2. 机器鱼ROS系统中的IMU订阅话题节点的代码：

订阅话题为：TransducerM

```imu_listener.py
#!/usr/bin/env python3
#!coding=utf-8

"""
    imu 数据保存节点
"""
import csv
import time
import rospy

from datetime import datetime
import os
from TransducerM_pkg.msg import Imu


class IMUSave():
    def __init__(self):
        rospy.loginfo("初始化IMU控制器!")
        
        self.count_max = rospy.get_param("~imu_nums_T", 100)
        self.enable_save = rospy.get_param('~enable_save', False)

        if type(self.enable_save) is not bool:
            rospy.logwarn("enable_save(true,false)参数类型错误，使用默认值False")
            self.enable_save = False
        

        if self.enable_save:
            # csv数据
            current_dir = os.path.dirname(os.path.abspath(__file__))
            # print("当前文件路径", current_dir)
            file_json_name = datetime.now().strftime('%Y_%m_%d_%H_%M_%S') + "imu_data.csv"
            self.csv_file = open(os.path.join(current_dir, '../../data/imu/', file_json_name), mode='w')
            self.csv_writer = csv.writer(self.csv_file)
            self.csv_writer.writerow(["time", "roll", "pitch", "yaw"])

        self.count_now = 0

        self.sub = rospy.Subscriber("TransducerM", Imu, self.mycontrol, queue_size=1)

        rospy.loginfo("初始化IMU控制器完成!")


    def mycontrol(self, msg): 
        # 保留原有的IMU读取和转换代码
        stamp = msg.header.stamp.to_sec()
        # 转化成弧度制
        roll = msg.roll * 3.1415926 / 180.0
        pitch = msg.pitch * 3.1415926 / 180.0
        yaw = msg.yaw * 3.1415926 / 180.0

        # rospy.loginfo("接收imu话题: 时间=%.4f RPY=[%.2f %.2f %.2f]", 
        #               msg.header.stamp.to_sec(), 
        #               msg.roll, msg.pitch, msg.yaw)
        # rospy.loginfo("转化成弧度制:%.4f %.4f %.4f", 
        #             msg.roll*3.1415926/180.0, 
        #             msg.pitch*3.1415926/180.0, 
        #             msg.yaw*3.1415926/180.0)

        if self.enable_save:
            self.csv_writer.writerow([stamp,roll, pitch, yaw])


        self.count_now += 1
        if(self.count_now % 50 == 0):
            rospy.loginfo(f"IMU数据 - 角度: RPY=[{msg.roll:.2f}, {msg.pitch:.2f}, {msg.yaw:.2f}], 弧度: [{msg.roll*3.1415926/180.0:.4f}, {msg.pitch*3.1415926/180.0:.4f}, {msg.yaw*3.1415926/180.0:.4f}]")

        if(self.count_now >= self.count_max):
            rospy.signal_shutdown("结束imu listener")    

if __name__ == "__main__":
    rospy.init_node("imu_listener")
    mycontrol = IMUSave()
    rospy.spin()
```

2. 机器鱼ROS系统中的UWB订阅话题节点的代码：

订阅话题为：/nlink_linktrack_nodeframe2

```uwb_listener.py
#!/usr/bin/env python3
#!coding=utf-8

"""
    uwb 测试和数据保存节点
"""

import csv
import time
import rospy

from datetime import datetime
import os
from nlink_parser.msg import LinktrackNodeframe2


class UWBSave():
    def __init__(self):
        rospy.loginfo("初始化UWB控制器!")
        
        self.count_max = rospy.get_param("~uwb_nums_T", 100)
        self.enable_save = rospy.get_param('~enable_save', False)

        if type(self.enable_save) is not bool:
            rospy.logwarn("enable_save(true,false)参数类型错误，使用默认值False")
            self.enable_save = False

        if self.enable_save:
            # csv数据
            current_dir = os.path.dirname(os.path.abspath(__file__))
            # print("当前文件路径", current_dir)
            file_json_name = datetime.now().strftime('%Y_%m_%d_%H_%M_%S') + "uwb_data.csv"
            self.csv_file = open(os.path.join(current_dir, '../../data/uwb/', file_json_name), mode='w')
            self.csv_writer = csv.writer(self.csv_file)
            self.csv_writer.writerow(["time", "x", "y", "z", "data_vel_3d_0", "data_vel_3d_1", "data_vel_3d_2"])

        self.count_now = 0

        self.sub = rospy.Subscriber("/nlink_linktrack_nodeframe2", LinktrackNodeframe2, self.mycontrol, queue_size=1)
        
        rospy.loginfo("初始化UWB控制器完成!")


    def mycontrol(self, msg): 

        if self.enable_save:
            self.csv_writer.writerow([time.time(), msg.pos_3d[0], msg.pos_3d[1], msg.pos_3d[2],
                                            msg.vel_3d[0], msg.vel_3d[1], msg.vel_3d[2]])

        self.count_now += 1
        if(self.count_now % 50 == 0):
            rospy.loginfo(f"位置xyz :{msg.pos_3d[0], msg.pos_3d[1], msg.pos_3d[2]}, 速度vxyz :{msg.vel_3d[0], msg.vel_3d[1], msg.vel_3d[2]}")

        if(self.count_now >= self.count_max):
            rospy.signal_shutdown("结束")    

if __name__ == "__main__":
    rospy.init_node("uwb_listener")
    mycontrol = UWBSave()
    rospy.spin() 

```

```wf5803_test.py
#!/usr/bin/env python3

'''
    测试深度传感器
    先启动WF5803_pub_node.py,发布深度传感器数据
    再启动wf5803_test.py,订阅深度传感器数据
'''

import rospy
from fish_control.msg import WF5803_MSG 


def wf5803_callback(msg):
    print(f"deep_raw_1: {msg.deep_raw_1:.4f}, deep_smooth_1: {msg.deep_smooth_1:.4f}, " + 
          f"deep_raw_2: {msg.deep_raw_2:.4f}, deep_smooth_2: {msg.deep_smooth_2:.4f}, " +
          f"deep_fusion: {msg.deep_fusion:.4f}")


if __name__ == "__main__":
    rospy.init_node("wf5803_test",anonymous=True)
    wf5803_sub = rospy.Subscriber("WF5803_topic",WF5803_MSG,wf5803_callback)
    rospy.spin()
```

# 4.鱼眼摄像头数据

在机器鱼里面只有一个保存视频文件的程序。

```save_video.py
#!/usr/bin/env python3
#!coding=utf-8

'''
 使用 python save_camera.py 运行
'''
import cv2
import time

# 0 表示第一个摄像头（可以改为 1,2... 测试其他设备）
cap = cv2.VideoCapture(0)

# 检查摄像头是否打开
if not cap.isOpened():
    print("× 无法打开摄像头")
    exit()

print("√ 摄像头打开成功")

# 获取图像宽高
frame_width = int(cap.get(3))
frame_height = int(cap.get(4))

# 设置视频编码器
fourcc = cv2.VideoWriter_fourcc(*'XVID') # 可以是'XVID'、'MJPG'、'DIVX'、'X264'、'H264'、'MPEG'、 'MP4V'

# 设置保存路径
timestamp = time.strftime("%Y%m%d_%H%M%S")
save_dir = "/home/orangepi/catkin_ws/src/fish_control/data/video/"
filename = f"{save_dir}video_{timestamp}.avi"

# 参数：文件名，编码器，帧率，分辨率
out = cv2.VideoWriter(filename, fourcc, 20.0, (frame_width, frame_height))

start_time = time.time()
duration = 5  # 录制时长10秒

while True:
    ret, frame = cap.read()
    if not ret:
        break
    out.write(frame)

    if time.time() - start_time > duration:
        print("录制结束")
        break

cap.release()
out.release()
```

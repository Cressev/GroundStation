这是机器鱼胸鳍和尾鳍两个舵机的使用说明文档。

之前我实现过一个用主机ssh连接机器鱼，并通过主机键盘控制机器鱼舵机运动的程序。

舵机控制话题：
- /servo_info_A：尾鳍幅值，单位：弧度，安全数据范围 0-0.4
- /servo_info_fre：尾鳍频率，单位：赫兹，安全数据范围 0-2.0
- /servo_info_bias：尾鳍偏置，单位：弧度，安全数据范围 -0.2-0.2
- /servo_info_angle：胸鳍转动角度，单位：微秒（pwm脉宽），安全数据范围 1300-2000

void ServoControl() {
    ser.setPort("/dev/ttyS3");   // 根据实际情况设置串口的port
    ser.setBaudrate(115200);
    ser.setTimeout(to);
    ser.setParity(pt);
    ser.setBytesize(bt);
    ser.setFlowcontrol(ft);
    ser.setStopbits(st);
    try
    {
        //打开串口
        ser.open();
    }
    catch(serial::IOException& e)
    {
        ROS_ERROR_STREAM("Unable to open port.");
        exit(1); //非正常退出程序；
    }
    //判断串口是否打开成功
    if(ser.isOpen())
    {
        ROS_INFO_STREAM("/dev/ttyS3 is opened.");
    }

    mid_servo_bias = -0.03;

    A = ros::param::param("tail_A", 0.0);
    // bias = ros::param::param("tail_bias", 0.0);
    bias = 0.0;
    mid_servo_bias = ros::param::param("tail_bias", 0.0);
    fre = ros::param::param("tail_frequency", 2.0);  // hs
    angle = ros::param::param("pec_angle", 1775.0);   // 500 - 2500  0 -180度
    // printf("A:%f,bias:%f\n",A,bias);

    // 等待 imu校准
    wait_imu_calib = ros::param::param("~wait_imu_calib", false);

    if (wait_imu_calib) {
        ROS_INFO("(from servo_control)等待 IMU 校准完成...");
        while (ros::ok()) {
            if (ros::param::param("/imu_calibration_done", false)) {
                break;
            }
            // ROS_INFO("等待 /imu_calibration_done == True...");
            sleep(0.5);
        }
        ROS_INFO("✅ IMU 校准完成，servo_control 节点启动！");
    }
    else{
        ROS_WARN("(from servo_control)未设置wait_imu_calib参数 或者 设置为 false，servo_control 节点立即启动！");
    }

    sleep(1); 

    servo_init();

    servo_time = ros::param::param("~servo_time", 20);
    //定时关闭线程
    timer = std::thread([]() {
        std::this_thread::sleep_for(std::chrono::seconds(servo_time));
        servo_close();
    });
    //发送线程
    send_thread = std::thread(send_task);
}
void test_handle(int signum){
    std::cout << "Interrupt signal (" << signum << ") received.\n";

void send_task() {
    while (ros::ok()) {
        // std::this_thread::sleep_for(std::chrono::microseconds(500));
        timingus(100);
        while (!msg_que.empty()) {
            std::lock_guard<std::mutex> lock(msg_mutex);
            auto item = msg_que.front();
            msg_que.pop();
            serial_send(item.cmd, item.data);
            // std::this_thread::sleep_for(std::chrono::microseconds(100));
            timingus(100);
        }
    }
}

void serial_send(uint8_t cmd, float data) {
    uint8_t serial_group[6];
    serial_group[0]=cmd;
    if(cmd==0xcc){
        if(data>=0){
            serial_group[1]=0;
        }else{
            serial_group[1]=1;
            data = -data;
        }
    }else{
        serial_group[1]=uint8_t(data/100);
    }
    serial_group[2]=uint8_t(int(data)%100);
    serial_group[3]=uint8_t(int(data*100)%100);
    serial_group[4]=uint8_t(int(data*10000)%100);
    serial_group[5]=0xff;

    int delay_time = 50; // us
    for(int i=0;i<6;i++){
        ser.write(&serial_group[i],1);
        usleep(delay_time);
    }
    // printf("send %d:%f\n",cmd,data);
}

void servo_init() {
    serial_send(0xcc, bias + mid_servo_bias);
    timingus(1000);

    serial_send(0xaa, A);
    timingus(1000);

    serial_send(0xbb, fre);
    timingus(1000);

    serial_send(0xdd, angle);
    timingus(1000);

    ROS_INFO("舵机初始化完成开始运动！");
    ROS_INFO_STREAM("初始参数 A: " << A << ", fre: " << fre << ", bias: " << bias << ", angle: " << angle);
}

void servo_close() {
    serial_send(0xaa, 0);
    timingus(5000);

    serial_send(0xbb, 0);
    timingus(5000);

    serial_send(0xcc, 0 + mid_servo_bias);
    timingus(5000);

    serial_send(0xdd, 0);
    timingus(5000);

    uint8_t cmd[6];
    cmd[0]=0xab;
    ser.write(&cmd[0],1);
    ros::shutdown();
    ROS_WARN("舵机结束运动！");
}

void signal_handler(int signal) {
    ROS_WARN("crtl c 舵机结束运动！");
    servo_close();
    exit(signal);  
    ros::shutdown();
}

void timingus(size_t us)
{
    auto start = std::chrono::system_clock::now();
    while (1) 
    {
        auto end = std::chrono::system_clock::now();
        auto duration = std::chrono::duration_cast<std::chrono::microseconds>(end - start);
        if (duration.count() > us) break;
    }
}

// 2024.08.28 yxq 键盘控制机器鱼游动，w前进，a左转，d右转，s停止
#include <ros/ros.h>
#include <std_msgs/Int32.h>
#include <std_msgs/Float32.h>

#include <termios.h>
#include <signal.h>
#include <math.h>

#define KEYCODE_W 0x77
#define KEYCODE_A 0x61
#define KEYCODE_S 0x73
#define KEYCODE_D 0x64

#define KEYCODE_A_CAP 0x41
#define KEYCODE_D_CAP 0x44
#define KEYCODE_S_CAP 0x53
#define KEYCODE_W_CAP 0x57

#define KEYCODE_1 0x31  // 尾鳍中位
#define KEYCODE_2 0x32  // 胸鳍测试
#define KEYCODE_3 0x33
#define KEYCODE_4 0x34
#define KEYCODE_5 0x35
#define KEYCODE_6 0x36
#define KEYCODE_7 0x37

// 定义一个类，用于控制智能车的键盘操作
class KeyboardTeleopNode
{
    private:

        // 定义消息，用于发布控制命令
        std_msgs::Int32 key;
        std_msgs::Float32 A;
        std_msgs::Float32 fre;
        std_msgs::Float32 bias;
        std_msgs::Float32 angle;

        // 定义一个NodeHandle类型的对象，用于与ROS进行通信
        ros::NodeHandle n_;
        // 定义一个Publisher类型的对象，用于发布控制命令
        ros::Publisher pub_key;
        ros::Publisher pub_A;
        ros::Publisher pub_fre;
        ros::Publisher pub_bias;
        ros::Publisher pub_angle;
    public:
        // 构造函数，初始化参数
        KeyboardTeleopNode()
        {
            pub_key = n_.advertise<std_msgs::Int32>("/key_board_pub", 1);
            pub_A = n_.advertise<std_msgs::Float32>("/servo_info_A", 1);
            pub_fre = n_.advertise<std_msgs::Float32>("/servo_info_fre", 1);
            pub_bias = n_.advertise<std_msgs::Float32>("/servo_info_bias", 1);
            pub_angle = n_.advertise<std_msgs::Float32>("/servo_info_angle", 1);
        }

        // 析构函数
        ~KeyboardTeleopNode() { }
        // 定义一个函数，用于键盘循环
        void keyboardLoop();
        // 消息发布函数
        void pub_msgs(float A,float fre,float bias,float angle);
        // 定义一个函数，用于停止机器人
        void stopRobot()
        {
            this->key.data = 0;
            this->A.data = 0.0;
            this->fre.data = 0.0;
            this->bias.data = 0.0;
            pub_key.publish(this->key);
            pub_A.publish(this->A);
            pub_fre.publish(this->fre);
            pub_bias.publish(this->bias);
        }
};


// 定义一个全局变量，用于存储键盘操作的实例
KeyboardTeleopNode* tbk;
// 定义一个文件描述符，用于存储键盘的文件描述符
int kfd = 0;
// 定义两个termios结构体，用于存储原始和修改后的终端设置
struct termios cooked, raw;
// 定义一个布尔变量，用于标记是否完成
bool done;


int main(int argc,char **argv)
{
    setlocale(LC_ALL,"");
    ros::init(argc,argv,"key_board");

    ros::NodeHandle nh_;
    // ros::Publisher pub = n.advertise<std_msgs::Float32>("/key_board_pub",1);


    // 创建一个键盘操作的实例
    KeyboardTeleopNode tbk;
    // 创建一个线程，用于执行键盘循环
    boost::thread t = boost::thread(boost::bind(&KeyboardTeleopNode::keyboardLoop, &tbk));

    // 进入ROS的事件循环
    ros::spin();

    // 中断线程
    t.interrupt();
    // 等待线程结束
    t.join();
    // 停止机器人
    tbk.stopRobot();
    // 恢复终端设置
    tcsetattr(kfd, TCSANOW, &cooked);

    return(0);
}

// 键盘循环函数
void KeyboardTeleopNode::keyboardLoop()
{
    // 定义一个字符变量，用于存储键盘输入
    char c;

    // 定义一个布尔变量，用于标记是否需要更新速度
    bool dirty = false;


    // 获取终端设置
    tcgetattr(kfd, &cooked);
    // 复制终端设置到raw结构体中
    memcpy(&raw, &cooked, sizeof(struct termios));
    // 设置终端为原始模式
    raw.c_lflag &=~ (ICANON | ECHO);
    raw.c_cc[VEOL] = 1;
    raw.c_cc[VEOF] = 2;
    tcsetattr(kfd, TCSANOW, &raw);

    puts("Reading from keyboard");
    puts("Use WASD keys to control the robot");
    // puts("Press Shift to move faster");

    // 定义一个pollfd结构体，用于存储文件描述符和事件
    struct pollfd ufd;
    ufd.fd = kfd;
    ufd.events = POLLIN;

    // 无限循环
    for(;;)
    {
        // 检查线程是否被中断
        boost::this_thread::interruption_point();

        // 从键盘读取下一个事件
        int num;

        // 调用poll函数，等待键盘输入
        if ((num = poll(&ufd, 1, 250)) < 0)
        {
            perror("poll():");
            return;
        }
        // 如果有键盘输入
        else if(num > 0)
        {
            // 读取键盘输入
            if(read(kfd, &c, 1) < 0)
            {
                perror("read():");
                return;
            }
        }
        // 如果没有键盘输入
        else
        {
            // 如果速度和转向有变化，则停止机器人
            if (dirty == true)
            {
                //stopRobot();
                dirty = false;
            }
            continue;
        }
        // 慢速行驶
        float a = 0.2;
        float fre = 2.0;
        float bias = 0.4;
        // 加速行驶
        float A = 0.3;
        float FRE = 2.2;
        float BIAS = 0.5;

        // 根据键盘输入，设置速度和转向
        switch(c)
        {
            case KEYCODE_W://前进
                this->key.data = 1;
                dirty = true;
                pub_msgs(a,fre,0,0);
                break;
            case KEYCODE_S://后退--停止
                this->key.data = 2;
                pub_msgs(0.0, 0.0, 0.0,0);
                dirty = true;
                break;
            case KEYCODE_A://左转
                this->key.data = 3;
                pub_msgs(a,fre, -bias,0);
                dirty = true;
                break;
            case KEYCODE_D://右转
                this->key.data = 4;
                pub_msgs(a,fre, bias,0);
                dirty = true;
                break;

            case KEYCODE_W_CAP://加速
                this->key.data = 5;
                pub_msgs(A,FRE,0.0,0);
                dirty = true;
                break;
            case KEYCODE_S_CAP://停止
                this->key.data = 6;
                pub_msgs(0.0,0.0,0.0,0);
                dirty = true;
                break;
            case KEYCODE_A_CAP://加速左转
                this->key.data = 7;
                pub_msgs(A,fre,-BIAS,0);
                dirty = true;
                break;
            case KEYCODE_D_CAP://加速右转
                this->key.data = 8;
                pub_msgs(A,fre,BIAS,0);
                dirty = true;
                break;
            case KEYCODE_1://尾鳍中位
                this->key.data = 9;
                pub_msgs(0.0,0.0,0.0,0.0);
                dirty = true;
                break;
            case KEYCODE_2://胸鳍测试1
                this->key.data = 10;
                pub_msgs(0.0,0.0,0.0,1500);
                dirty = true;
                break;
            case KEYCODE_3://胸鳍测试2
                this->key.data = 10;
                pub_msgs(0.0,0.0,0.0,1800);
                dirty = true;
                break;
            default://无按键则停止
                this->key.data = 0;
                //pub_msgs(0.0, 0.0, 0.0);
                dirty = false;
        }

        // 发布控制命令
        // pub_.publish(key);
        ROS_INFO("key:%d", this->key.data);
    }
}

void KeyboardTeleopNode::pub_msgs(float A,float fre,float bias,float angle)
{
    this->A.data = A;
    this->fre.data = fre;
    this->bias.data = bias;
    this->angle.data = angle;
    pub_A.publish(this->A);
    pub_fre.publish(this->fre);
    pub_bias.publish(this->bias);
    pub_angle.publish(this->angle);
}
